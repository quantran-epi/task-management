**Bounded 1-shot repair pattern (D-16):**
```typescript
// First attempt
const initial = await this.executeCompletion(messages, options);
const check = this.validatePlan(initial, context);
if (check.valid && check.plan) {
  return { type: 'plan', plan: check.plan, repaired: false };
}

// Exactly one repair attempt
const repairMessages = [
  ...messages,
  { role: 'assistant' as const, content: JSON.stringify(initial) },
  {
    role: 'user' as const,
    content: `Repair this logical plan. Errors: ${check.issues.join('; ')}. Output only valid JSON.`,
  },
];
const repaired = await this.executeCompletion(repairMessages, options);
const repairCheck = this.validatePlan(repaired, context);
if (repairCheck.valid && repairCheck.plan) {
  return { type: 'plan', plan: repairCheck.plan, repaired: true };
}

return {
  type: 'unsupported',
  reason: 'PLANNER_SCHEMA_FAILURE',
  unsupportedConstructs: check.issues,
};
```

### Strict Hostname Validation
**Source:** `knowledge-server/src/main.ts` (lines 58-72)  
**Apply to:** Remote 9router endpoint validation
```typescript
function parseAllowedHosts(rawHosts: string, varName: string): string[] {
  const hosts = rawHosts.split(',').map((h) => h.trim()).filter(Boolean);
  if (hosts.length === 0) throw new Error(`${varName} must contain at least one valid host.`);
  for (const host of hosts) {
    if (host.includes('/') || host.includes(':') || host.includes('@') || host.includes('*')) {
      throw new Error(`Invalid host in ${varName}: "${host}". Must be exact hostname without port, path, or wildcards.`);
    }
  }
  return hosts;
}
```
