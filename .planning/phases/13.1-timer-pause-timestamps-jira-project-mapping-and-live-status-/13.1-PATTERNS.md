### 3. High-Fidelity Timer Segments & Safe Automation

**Segment manipulation & status automation pattern:**

// Pause: Close current open segment
const pauseTimer = useCallback(async (taskId: string) => {
  const timer = await database.activeTimers.get(taskId);
  if (!timer || timer.status !== 'running') return;

  const nowIso = new Date().toISOString();
  const currentRunMs = Math.max(0, Date.now() - timer.startedAt);
  const updatedSegments = [...(timer.segments || [])];
  
  if (updatedSegments.length > 0 && !updatedSegments[updatedSegments.length - 1]!.endTime) {
    updatedSegments[updatedSegments.length - 1]!.endTime = nowIso;
  }

  await database.activeTimers.put({
    ...timer,
    status: 'paused',
    accumulatedMs: timer.accumulatedMs + currentRunMs,
    segments: updatedSegments,
  });
  broadcastTimerSync('pause', taskId);
}, [database]);

// Resume: Push new open segment + Safe status automation (Open -> In Progress only)
const startTimer = useCallback(async (taskId: string) => {
  const nowIso = new Date().toISOString();
  const nowMs = Date.now();

  await database.transaction('rw', [database.activeTimers, database.tasks], async () => {
    const task = await database.tasks.get(taskId);
    if (task && task.status === 'Open') {
      task.status = 'In Progress';
      task.updatedAt = nowIso;
      await database.tasks.put(task);
    }

    const existing = await database.activeTimers.get(taskId);
    if (existing) {
      const updatedSegments = [...(existing.segments || []), { startTime: nowIso }];
      await database.activeTimers.put({
        ...existing,
        status: 'running',
        startedAt: nowMs,
        segments: updatedSegments,
      });
    } else {
      await database.activeTimers.put({
        taskId,
        status: 'running',
        startedAt: nowMs,
        accumulatedMs: 0,
        sessionStartTime: nowIso,
        segments: [{ startTime: nowIso }],
      });
    }
  });
  broadcastTimerSync('start', taskId);
}, [database]);

// Finish while paused: Use last pause timestamp as endTime (D-03)
const finishTimer = useCallback(async (taskId: string, note?: string) => {
  const timer = await database.activeTimers.get(taskId);
  if (!timer) return null;

  const nowIso = new Date().toISOString();
  const lastSegment = timer.segments?.[timer.segments.length - 1];
  const endTime = timer.status === 'paused' && lastSegment?.endTime
    ? lastSegment.endTime
    : nowIso;

  // Aggregate segments into WorkSession
  const session = await createWorkSession({
    taskId,
    startTime: timer.sessionStartTime,
    endTime,
    durationMinutes: Math.max(1, Math.round(timer.accumulatedMs / 60000)),
    segments: timer.segments,
    note,
  }, database);

  await database.activeTimers.delete(taskId);
  broadcastTimerSync('finish', taskId);
  return session;
}, [database]);
```

// src/utils/worklog.ts
export function splitSegmentByMidnight(
  startIso: string,
  endIso: string
): Array<{ date: string; minutes: number }> {
  const start = new Date(startIso);
  const end = new Date(endIso);
  const results: Array<{ date: string; minutes: number }> = [];

  let current = new Date(start);
  while (current < end) {
    const nextMidnight = new Date(current.getFullYear(), current.getMonth(), current.getDate() + 1);
    const segmentEnd = end < nextMidnight ? end : nextMidnight;
    const diffMinutes = Math.max(1, Math.round((segmentEnd.getTime() - current.getTime()) / 60000));
    
    const year = current.getFullYear();
    const month = String(current.getMonth() + 1).padStart(2, '0');
    const day = String(current.getDate()).padStart(2, '0');
    results.push({ date: `${year}-${month}-${day}`, minutes: diffMinutes });

    current = segmentEnd;
  }
  return results;
}
```

// Backoff ladder: 1 -> 5 -> 15 -> 30 -> 60 minutes
export const BACKOFF_MINUTES = [1, 5, 15, 30, 60] as const;

export function getNextBackoffDelayMinutes(attemptCount: number): number {
  const idx = Math.min(attemptCount, BACKOFF_MINUTES.length - 1);
  return BACKOFF_MINUTES[idx]!;
}

// In sync error handler:
if (err instanceof GitHubSyncConflictError) {
  // Remote SHA mismatch -> pause auto-sync scheduler immediately (D-42)
  await db.settings.put({
    key: 'github_auto_sync_state',
    value: { isPaused: true, reason: 'conflict', conflictSha: err.remoteSha }
  });
} else if (isAuthError(err)) {
  // 401/403 -> stop retry, prompt Replace Token (D-39)
  await db.settings.put({
    key: 'github_auto_sync_state',
    value: { isPaused: true, reason: 'auth_error' }
  });
} else {
  // Network/5xx -> schedule exponential backoff
  const nextRetryMs = Date.now() + getNextBackoffDelayMinutes(currentAttempts) * 60000;
  await db.settings.put({
    key: 'github_auto_sync_state',
    value: { nextRetryAt: new Date(nextRetryMs).toISOString(), attempts: currentAttempts + 1 }
  });
}
