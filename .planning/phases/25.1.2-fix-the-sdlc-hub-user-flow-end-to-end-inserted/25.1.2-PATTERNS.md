```typescript
export const Step2SolutionView: React.FC<{
  projectId: string;
  solutionArtifact?: SdlcArtifactRef;
  aiSuggestedDraft?: string;
  onCommitSolution: (markdown: string, noteId: string) => Promise<void>;
}> = ({ projectId, solutionArtifact, aiSuggestedDraft, onCommitSolution }) => {
  const [content, setContent] = useState<string>(solutionArtifact?.content ?? aiSuggestedDraft ?? '');
  const [mode, setMode] = useState<'write' | 'preview' | 'split'>('write');
  const [saving, setSaving] = useState(false);

  // Update content from AI stream only if user hasn't edited or explicitly clicks apply
  const handleApplyAiDraft = (draft: string) => {
    setContent(draft);
    message.info('Đã áp dụng bản thảo từ AI');
  };

  const handleApproveAndSave = async () => {
    if (!content.trim()) {
      message.error('Nội dung SOLUTION.md không được để trống');
      return;
    }
    setSaving(true);
    try {
      const result = await saveHubSolution({
        db,
        projectId,
        threadId: `sdlc:${projectId}`,
        sessionId: crypto.randomUUID(),
        markdown: content,
        draftRevision: (solutionArtifact?.version ?? 0) + 1,
      });
      await onCommitSolution(content, result.noteId);
      message.success('Đã duyệt và lưu SOLUTION.md');
    } catch (err: any) {
      message.error(err.message || 'Lưu thất bại');
    } finally {
      setSaving(false);
    }
  };

  return (
    <div style={{ display: 'flex', flexDirection: 'column', height: '100%', gap: 12 }}>
      <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center' }}>
        <Segmented
          value={mode}
          onChange={(val) => setMode(val as any)}
          options={[
            { label: 'Soạn thảo', value: 'write', icon: <EditOutlined /> },
            { label: 'Xem trước', value: 'preview', icon: <EyeOutlined /> },
            { label: 'Chia đôi', value: 'split', icon: <SplitCellsOutlined /> },
          ]}
        />
        <Button type="primary" icon={<CheckOutlined />} loading={saving} onClick={handleApproveAndSave}>
          Duyệt & Lưu SOLUTION.md
        </Button>
      </div>

      <div style={{ flex: 1, display: 'flex', gap: 12, overflow: 'hidden' }}>
        {(mode === 'write' || mode === 'split') && (
          <Input.TextArea
            value={content}
            onChange={(e) => setContent(e.target.value)}
            style={{ flex: 1, resize: 'none', height: '100%', fontFamily: 'monospace' }}
            placeholder="Nội dung kiến trúc giải pháp SOLUTION.md..."
          />
        )}
        {(mode === 'preview' || mode === 'split') && (
          <div
            style={{ flex: 1, overflowY: 'auto', padding: 12, border: '1px solid #d9d9d9', borderRadius: 6 }}
            dangerouslySetInnerHTML={{ __html: renderSafeMarkdown(content) }}
          />
        )}
      </div>
    </div>
  );
};
```
