interface SdlcWrapperPromptInput {
  userQuery: string;
  step?: string | undefined;
  metadata?: { cr?: string; featureName?: string; requestingUnit?: string } | undefined;
  artifacts?: { pl01?: ...; pl02?: ...; solution?: ... } | undefined;
}
export function buildSdlcWrapperPrompt(input: SdlcWrapperPromptInput): string {
  const parts: string[] = [];
  parts.push(`[Quy trình SDLC - Bước: ${step}]`);
  if (input.metadata) { /* cr / featureName / requestingUnit */ }
  parts.push(`\n[CHÍNH SÁCH BẮT BUỘC CHẾ ĐỘ PHÂN TÍCH (ANALYSIS MODE)]: ...`);
  if (input.artifacts?.pl01?.content) parts.push(`\n[YÊU CẦU NGHIỆP VỤ (PL01)]${refs}:\n${...}`);
  if (input.artifacts?.pl02?.content) parts.push(`\n[ĐẶC TẢ CHỨC NĂNG (PL02)]${refs}:\n${...}`);
