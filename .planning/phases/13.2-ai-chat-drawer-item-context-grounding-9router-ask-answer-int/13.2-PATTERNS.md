export async function getNineRouterApiKey(db: TaskPlannerDatabase = defaultDb): Promise<string> {
  if (memoryToken !== null) {
    return memoryToken;
  }

  if (isTauriApp()) {
    const token = await getKeyringCredential(KEYRING_KEYS.NINEROUTER_API_KEY);

    memoryToken = token;
  }
  return token;
}

export async function setNineRouterApiKey(
  token: string,
  db: TaskPlannerDatabase = defaultDb
): Promise<void> {
  const trimmed = token.trim();
  memoryToken = trimmed;

  if (isTauriApp()) {
    if (trimmed) {
      await storeKeyringCredential(KEYRING_KEYS.NINEROUTER_API_KEY, trimmed);
    } else {
      await deleteKeyringCredential(KEYRING_KEYS.NINEROUTER_API_KEY);
    }
    await db.settings.delete('ninerouter_api_key');
  } else {
    if (trimmed) {
      await db.settings.put({ key: 'ninerouter_api_key', value: trimmed });
    } else {
      await db.settings.delete('ninerouter_api_key');
    }
  }
}

**ReadableStream Line-Buffering SSE Generator Pattern** (standard Web Streams API):
```typescript
export async function* streamChatCompletion(
  endpoint: string,
  apiKey: string,
  payload: { model: string; messages: Array<{ role: string; content: string }>; temperature?: number },
  signal?: AbortSignal
): AsyncGenerator<string, void, unknown> {
  const url = `${endpoint.replace(/\/+$/, '')}/chat/completions`;
  const response = await fetch(url, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${apiKey}`,
    },
    body: JSON.stringify({ ...payload, stream: true }),
    signal,
  });

  if (!response.ok) {
    const errorText = await response.text();
    throw new Error(`API Error (${response.status}): ${sanitizeErrorMessage(errorText, apiKey)}`);
  }

  if (!response.body) {
    throw new Error('Response body is empty');
  }

  const reader = response.body.getReader();
  const decoder = new TextDecoder('utf-8');
  let lineBuffer = '';

  try {
    while (true) {
      const { done, value } = await reader.read();
      if (done) break;

      lineBuffer += decoder.decode(value, { stream: true });
      const lines = lineBuffer.split(/\r?\n/);
      lineBuffer = lines.pop() ?? '';

      for (const line of lines) {
        const trimmed = line.trim();
        if (!trimmed || trimmed.startsWith(':')) continue;

        if (trimmed.startsWith('data:')) {
          const dataContent = trimmed.slice(5).trim();
          if (dataContent === '[DONE]') return;

          try {
            const parsed = JSON.parse(dataContent);
            const deltaText = parsed.choices?.[0]?.delta?.content;
            if (deltaText) yield deltaText;
          } catch {
            // Incomplete JSON segment across chunk split
          }
        }
      }
    }
  } finally {
    reader.releaseLock();
  }
}
```

import { db as defaultDb, type TaskPlannerDatabase } from '../index';
import { generateId } from '../../utils/uuid';
import type { ChatThread, ChatMessage, ScopeType } from '../schema';

export async function getOrCreateThreadForScope(
  scopeType: ScopeType,
  scopeId: string,
  db: TaskPlannerDatabase = defaultDb
): Promise<ChatThread> {
  const scopeKey = `${scopeType}:${scopeId}`;
  const existing = await db.chatThreads.where('scopeKey').equals(scopeKey).first();
  if (existing) return existing;

  const now = new Date().toISOString();
  const newThread: ChatThread = {
    id: generateId(),
    scopeKey,
    scopeType,
    scopeId,
    createdAt: now,
    updatedAt: now,
  };
  await db.chatThreads.add(newThread);
  return newThread;
}

export async function appendChatMessage(
  threadId: string,
  role: 'user' | 'assistant' | 'system' | 'divider',
  content: string,
  db: TaskPlannerDatabase = defaultDb
): Promise<ChatMessage> {
  const now = new Date().toISOString();
  const message: ChatMessage = {
    id: generateId(),
    threadId,
    role,
    content,
    createdAt: now,
  };
  await db.chatMessages.add(message);
  await db.chatThreads.update(threadId, { updatedAt: now });
  return message;
}

async function tauriInvoke<T>(command: string, args?: Record<string, unknown>): Promise<T> {
  const api = await import('@tauri-apps/api/core');
  return args === undefined ? api.invoke<T>(command) : api.invoke<T>(command, args);
}

export function buildClaudeCliCommand(taskName: string, promptContent: string): string {
  const safeTitle = taskName.replace(/["\\]/g, '');
  const safeContent = promptContent.replace(/["\\]/g, ' ');
  return `claude "${safeTitle}: ${safeContent}"`;
}

export async function launchClaudeCodeTerminal(commandStr: string): Promise<void> {
  if (isTauriApp()) {
    try {
      await tauriInvoke('launch_claude_terminal', { commandStr });
      message.success('Đã mở Claude Code trong Terminal');
      return;
    } catch (err: any) {
      message.error(`Không thể mở Terminal: ${err?.message || err}`);
    }
  }

  // Web/PWA fallback: Copy CLI command to clipboard
  try {
    if (navigator?.clipboard?.writeText) {
      await navigator.clipboard.writeText(commandStr);
      message.info('Đã sao chép lệnh Claude Code vào clipboard');
    }
  } catch {
    message.warning('Không thể sao chép lệnh vào clipboard');
  }
}

## Shared Patterns

### Dual-Tier Credential Pattern
**Source:** `src/services/jiraTokenService.ts`  
**Apply to:** `src/services/ai/nineRouterTokenService.ts`  
- Check `isTauriApp()` first.
- In desktop Tauri: invoke `keyring_store` (`store_credential`, `get_credential`, `delete_credential`) under service `task-planner`.
- In Web/PWA: read and write to `db.settings` table.
- Cache in module-level `let memoryToken: string | null` for instant UI access.

### Cascade Atomic Deletion Pattern
**Source:** `src/db/repositories/cascadeRepo.ts`  
**Apply to:** `deleteProjectWithCascade`, `deleteMilestoneWithCascade`, `deleteTaskWithAllocations`  
- Enclose all deletion steps in a single atomic `db.transaction('rw', [...stores], async () => { ... })`.
- Whenever an entity is deleted, query `db.chatThreads.where('scopeKey').equals(key)` and delete corresponding `chatMessages` and `chatThreads`.

### Input Validation & Safety Sanitization Pattern
**Source:** `src/utils/markdown.ts` & `src/services/jira/jiraApi.ts`  
**Apply to:** `contextGrounding.ts`, `nineRouterClient.ts`, `ChatMessageBubble.tsx`  
- All prompt parameters sanitized against prompt leakage.
- Passwords and secret API keys masked with `sanitizeErrorMessage` in any thrown error logs.
- AI assistant messages rendered using HTML escaping prior to Markdown parsing (`renderSafeMarkdown`).
