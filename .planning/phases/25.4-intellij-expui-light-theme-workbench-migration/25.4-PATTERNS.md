export const EditorArea: React.FC = () => {
  const { openTabs, activeTabId, selectTab, closeTab } = useLayoutStore();

  const renderActiveView = (tab: WorkbenchTab) => {
    switch (tab.route) {
      case 'tasks':
        return <TasksView />;
      case 'notes':
        return <NotesView initialDocId={tab.params?.doc} />;
      case 'planner':
        return <PlannerView targetDate={tab.params?.date} />;
      case 'projects':
        return <ProjectsView />;
      case 'analytics':
        return <AnalyticsView />;
      case 'settings':
        return <SettingsView />;
      case 'dashboard':
      default:
        return <DashboardView />;
    }
  };

  const activeTab = openTabs.find((t) => t.id === activeTabId) ?? openTabs[0];

  return (
    <div className="workbench-editor-container">
      <div role="tablist" className="workbench-editor-tabbar">
        {openTabs.map((tab) => {
          const isActive = tab.id === activeTabId;
          return (
            <div
              key={tab.id}
              role="tab"
              aria-selected={isActive}
              className={`workbench-editor-tab ${isActive ? 'is-active' : ''}`}
              onClick={() => selectTab(tab.id)}
            >
              <span className="workbench-tab-title">{tab.title}</span>
              {openTabs.length > 1 && (
                <button
                  type="button"
                  className="workbench-tab-close"
                  onClick={(e) => {
                    e.stopPropagation();
                    closeTab(tab.id);
                  }}
                  aria-label={`Đóng tab ${tab.title}`}
                >
                  <CloseOutlined style={{ fontSize: 10 }} />
                </button>
              )}
            </div>
          );
        })}
      </div>
      <div className="workbench-editor-canvas">
        {activeTab ? renderActiveView(activeTab) : null}
      </div>
    </div>

export function useTauriWindow() {
  const isTauri = isTauriApp();

  const minimize = async () => {
    if (!isTauri) return;
    try {
      const { getCurrentWebviewWindow } = await import('@tauri-apps/api/webviewWindow');
      await getCurrentWebviewWindow().minimize();
    } catch (err) {
      console.warn('Failed to minimize window:', err);
    }
  };

  const toggleMaximize = async () => {
    if (!isTauri) return;
    try {
      const { getCurrentWebviewWindow } = await import('@tauri-apps/api/webviewWindow');
      await getCurrentWebviewWindow().toggleMaximize();
    } catch (err) {
      console.warn('Failed to toggle maximize window:', err);
    }
  };

  const close = async () => {
    if (!isTauri) return;
    try {
      const { getCurrentWebviewWindow } = await import('@tauri-apps/api/webviewWindow');
      await getCurrentWebviewWindow().close();
    } catch (err) {
      console.warn('Failed to close window:', err);
    }
  };

  return { isTauri, minimize, toggleMaximize, close };
}

### `src/workbench/hooks/useDoubleShift.ts` (hook, event-driven)

**Analog:** `src/hooks/useGlobalShortcuts.ts` (lines 28-31)

**Core double-shift filter pattern**:
```typescript
import { useEffect, useRef } from 'react';

export function useDoubleShift(onDoubleShift: () => void) {
  const lastShiftRef = useRef<number>(0);
  const otherKeyPressedRef = useRef<boolean>(false);

  useEffect(() => {
    const handleKeyDown = (e: KeyboardEvent) => {
      if (e.key !== 'Shift') {
        otherKeyPressedRef.current = true;
      }
    };

    const handleKeyUp = (e: KeyboardEvent) => {
      if (e.key === 'Shift') {
        if (!otherKeyPressedRef.current) {
          const now = performance.now();
          if (now - lastShiftRef.current < 350) {
            onDoubleShift();
            lastShiftRef.current = 0;
          } else {
            lastShiftRef.current = now;
          }
        }
        otherKeyPressedRef.current = false;
      } else {
        otherKeyPressedRef.current = false;
        lastShiftRef.current = 0;
      }
    };

    window.addEventListener('keydown', handleKeyDown);
    window.addEventListener('keyup', handleKeyUp);
    return () => {
      window.removeEventListener('keydown', handleKeyDown);
      window.removeEventListener('keyup', handleKeyUp);
    };
  }, [onDoubleShift]);
}
```

export interface WorkbenchAction {
  id: string;
  category: 'navigation' | 'toolwindow' | 'action';
  title: string;
  shortcut?: string;
  keywords: string[];
  run: () => void | Promise<void>;
}

class ActionRegistry {
  private actions = new Map<string, WorkbenchAction>();

  register(action: WorkbenchAction): () => void {
    this.actions.set(action.id, action);
    return () => {
      this.actions.delete(action.id);
    };
  }

  getAll(): WorkbenchAction[] {
    return Array.from(this.actions.values());
  }

  get(id: string): WorkbenchAction | undefined {
    return this.actions.get(id);
  }

  execute(id: string): void {
    const action = this.actions.get(id);
    if (action) {
      void action.run();
    }
  }
}

export const actionRegistry = new ActionRegistry();
