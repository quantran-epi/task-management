```typescript
this.version(6)
  .stores(SCHEMA_V6)
  .upgrade(async (tx) => {
    const migrateRecordReminders = (record: any) => {
      if (!Array.isArray(record.reminders)) {
        record.reminders = [];
        if (record.reminderDate) {
          record.reminders.push({

            date: record.reminderDate,
            note: record.reminderNote || undefined,
          });
        }
      }

    await tx.table('milestones').toCollection().modify(migrateRecordReminders);
    await tx.table('tasks').toCollection().modify(migrateRecordReminders);
```

```typescript
// src/utils/notifications.ts
export function isReminderTriggered(
  reminder: ReminderItem,
  todayDate: string,
  currentTime: string // 'HH:mm'
): boolean {
  if (!reminder.date) return false;
  if (reminder.date < todayDate) return true; // Overdue reminder stays active
  if (reminder.date === todayDate) {
    if (!reminder.time) return true; // Triggered from start of day
    return currentTime >= reminder.time; // Triggered once target time reached
  }
  return false;
}

// In evaluateNotifications loop:
for (const reminder of (task.reminders || [])) {
  if (isReminderTriggered(reminder, todayDate, currentTime)) {
    const alertKey = `reminder:task:${task.id}:${reminder.id}`;
    if (dismissedMap[alertKey] !== todayDate) {
      results.push({
        id: alertKey,
        category: 'reminder',
        title: task.name,
        subtitle: reminder.note || (task.projectId ? projectMap.get(task.projectId) : undefined),
        date: reminder.time ? `${reminder.time} ${reminder.date}` : reminder.date,
        tagColor: 'gold',
        tagLabel: reminder.time ? `Nhắc nhở (${reminder.time})` : 'Nhắc nhở',
        entityType: 'task',
        entityId: task.id,
        canDismiss: true,
        priorityOrder: 5,
        task,
      });
    }
  }
}
```
