export function extractSolutionMarkdown(rawResponse: string): string {
  if (!rawResponse) return '';
  const clean = stripThinkingBlocks(rawResponse).trim();

  // 1. Look for fenced markdown code block containing SOLUTION heading
  const fencedMatch = clean.match(/```(?:markdown|md)?\s*([\s\S]*?)```/i);
  if (fencedMatch && (fencedMatch[1]?.includes('# SOLUTION') || fencedMatch[1]?.includes('## 1.'))) {
    return fencedMatch[1].trim();
  }

  // 2. Look for explicit # SOLUTION heading starting point
  const solutionIdx = clean.indexOf('# SOLUTION');
  if (solutionIdx !== -1) {
    return clean.slice(solutionIdx).trim();
  }

  // 3. Fallback: clean response
  return clean;
}
```

export async function dispatchAiToolCall(
  name: string,
  args: Record<string, any>,
  db: TaskPlannerDatabase,
  signal?: AbortSignal,
  options?: { allowMutation?: boolean }
): Promise<string> {
  const normalized = normalizeToolName(name);
  const isLocalDbTool = AI_DATABASE_TOOLS.some(
    (t) => normalizeToolName(t.function.name) === normalized
  );
  return isLocalDbTool
    ? executeAiTool(name, args, db, signal)
    : executeDynamicMcpTool(normalized, args, db, { allowMutation: options?.allowMutation });
}
```
